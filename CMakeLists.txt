cmake_minimum_required(VERSION 3.16)
project(funny_furries VERSION 0.6.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(FF_SANITIZE "Enable undefined behavior sanitizer" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
pkg_check_modules(SDL2_MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
add_library(ff_core src/game.c src/arena.c src/mod.c src/rollback.c)
target_include_directories(ff_core PUBLIC src)
target_compile_options(ff_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(FF_SANITIZE)
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all)
  add_link_options(-fsanitize=undefined)
  target_compile_options(ff_core PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
endif()
add_library(ff_net src/net.c src/netplay.c)
target_link_libraries(ff_net PUBLIC ff_core)
target_compile_options(ff_net PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(WIN32)
  target_link_libraries(ff_net PUBLIC ws2_32)
endif()
# Warpzone: a pinned libwarpzone release, checked against its SHA-256 like any
# other toolchain download. -DFF_WARPZONE=OFF builds without online rooms.
option(FF_WARPZONE "Online rooms and ranked play through Warpzone" ON)
set(FF_WARPZONE_DIR "" CACHE PATH "An unpacked libwarpzone package to use instead of downloading one")
add_library(ff_wz src/wz.c)
target_link_libraries(ff_wz PUBLIC ff_net)
target_compile_options(ff_wz PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(FF_WARPZONE)
  set(WZ_VERSION 0.1.0)
  if(WIN32)
    set(WZ_PACKAGE libwarpzone-${WZ_VERSION}-x86_64-windows-gnu)
    set(WZ_FILE ${WZ_PACKAGE}.zip)
    set(WZ_SHA256 6d39bd8679141562f1f9155669b5a1b408dad2be27cecdd96fb4f5b35a2b21e5)
  else()
    set(WZ_PACKAGE libwarpzone-${WZ_VERSION}-x86_64-linux-gnu)
    set(WZ_FILE ${WZ_PACKAGE}.tar.gz)
    set(WZ_SHA256 2881f7385fb2112cc6727f459834266f09b0f3cbdc92b0960aa4f9dcc2b51410)
  endif()
  if(NOT FF_WARPZONE_DIR)
    set(WZ_DEPS ${CMAKE_BINARY_DIR}/_deps)
    if(NOT EXISTS ${WZ_DEPS}/${WZ_PACKAGE}/include/warpzone.h)
      file(DOWNLOAD https://github.com/jarvis-jhk/warpzone/releases/download/v${WZ_VERSION}/${WZ_FILE} ${WZ_DEPS}/${WZ_FILE}
           EXPECTED_HASH SHA256=${WZ_SHA256} TLS_VERIFY ON)
      execute_process(COMMAND ${CMAKE_COMMAND} -E tar xf ${WZ_DEPS}/${WZ_FILE} WORKING_DIRECTORY ${WZ_DEPS} RESULT_VARIABLE WZ_UNPACKED)
      if(NOT WZ_UNPACKED EQUAL 0)
        message(FATAL_ERROR "Cannot unpack ${WZ_FILE}")
      endif()
    endif()
    set(FF_WARPZONE_DIR ${WZ_DEPS}/${WZ_PACKAGE})
  endif()
  target_compile_definitions(ff_wz PRIVATE FF_WARPZONE)
  target_include_directories(ff_wz PRIVATE ${FF_WARPZONE_DIR}/include)
  target_link_libraries(ff_wz PUBLIC ${FF_WARPZONE_DIR}/lib/libwarpzone.a)
  if(WIN32)
    target_link_libraries(ff_wz PUBLIC kernel32 advapi32 ntdll userenv ws2_32 bcrypt dbghelp)
  else()
    target_link_libraries(ff_wz PUBLIC pthread dl m)
  endif()
endif()
add_executable(funny-furries src/main.c src/input.c src/fault.c)
target_link_libraries(funny-furries PRIVATE ff_core ff_net ff_wz PkgConfig::SDL2 PkgConfig::SDL2_IMAGE PkgConfig::SDL2_MIXER)
if(NOT MSVC)
  target_link_libraries(funny-furries PRIVATE m)
endif()
target_compile_options(funny-furries PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
