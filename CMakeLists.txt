cmake_minimum_required(VERSION 3.16)
project(funny_furries VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(FF_SANITIZE "Enable undefined behavior sanitizer" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
pkg_check_modules(SDL2_MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
add_library(ff_core src/game.c src/arena.c)
target_include_directories(ff_core PUBLIC src)
target_compile_options(ff_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(FF_SANITIZE)
  add_compile_options(-fsanitize=undefined -fno-sanitize-recover=all)
  add_link_options(-fsanitize=undefined)
  target_compile_options(ff_core PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
endif()
add_executable(funny-furries src/main.c src/input.c src/fault.c)
target_link_libraries(funny-furries PRIVATE ff_core PkgConfig::SDL2 PkgConfig::SDL2_IMAGE PkgConfig::SDL2_MIXER)
target_compile_options(funny-furries PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_game tests/engine/test_game.c)
  target_link_libraries(test_game PRIVATE ff_core)
  target_compile_options(test_game PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME match COMMAND test_game)
  add_executable(test_input tests/engine/test_input.c src/input.c)
  target_link_libraries(test_input PRIVATE PkgConfig::SDL2)
  target_include_directories(test_input PRIVATE src)
  target_compile_options(test_input PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME input COMMAND test_input)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(fault_probe tests/engine/fault_probe.c src/fault.c)
  target_include_directories(fault_probe PRIVATE src)
  if(UNIX)
    add_test(NAME fault_courier COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/engine/test_fault.py $<TARGET_FILE:funny-furries> $<TARGET_FILE:fault_probe>)
  endif()
endif()
